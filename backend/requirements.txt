# ml/ package deps (training + inference).
# Pinned EXACTLY to the versions ml/artifacts/*.joblib were pickled with.
# joblib.load() unpickles scikit-learn estimators by resolving their
# internal module paths (e.g. sklearn's private `_loss` module) -- those
# paths change between minor versions, so an open-ended `>=` here lets a
# routine redeploy silently resolve a newer, incompatible scikit-learn and
# break every route at import time (ModuleNotFoundError: No module named
# '_loss'). Bump these only together with a retrain + artifact commit.
pandas==2.2.3
numpy==1.26.4
scikit-learn==1.5.1
joblib==1.5.1

# unpickling-compatibility reason as above.
